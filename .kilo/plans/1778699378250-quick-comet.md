# Universal Media Handoff System - Implementation Plan

## Overview

Transform the Higgsfield-style AI video platform into a FULLY CONNECTED MULTI-APP VIDEO ECOSYSTEM where all generated assets automatically flow between editors, directors, and AI agents.

## Architecture

```
Generated Asset
    ↓
Shared Asset Pipeline (src/lib/assets/)
    ↓
Central Asset Registry (localStorage/IndexedDB)
    ↓
Universal Handoff Actions
    ↓
Any Editor / Timeline / AI Agent
```

### 1.2 assetStore.js - Central Asset Registry

```javascript
import { ASSET_TYPES, ASSET_STATUS } from './assetSchema.js';

const STORAGE_KEYS = {
  ASSETS: 'universal_assets',
  METADATA: 'universal_assets_meta'
};

const defaultStorage = {
  getItem: (key) => localStorage.getItem(key),
  setItem: (key, value) => localStorage.setItem(key, value),
  removeItem: (key) => localStorage.removeItem(key)
};

export class AssetStore {
  constructor(options = {}) {
    this.storage = options.storage || defaultStorage;
    this.db = null;
    this.useIndexedDB = options.useIndexedDB || false;
    this.subscribers = new Set();
  }

  async init() {
    if (this.useIndexedDB && window.indexedDB) {
      await this.initIndexedDB();
    }
  }

  async initIndexedDB() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open('UniversalAssetStore', 1);
      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        this.db = request.result;
        resolve();
      };
      request.onupgradeneeded = (event) => {
        const db = event.target.result;
        const store = db.createObjectStore('assets', { keyPath: 'id' });
        store.createIndex('type', 'type', { unique: false });
        store.createIndex('sourceApp', 'sourceApp', { unique: false });
        store.createIndex('createdAt', 'createdAt', { unique: false });
      };
    });
  }

  async saveAsset(asset) {
    asset.updatedAt = new Date().toISOString();
    
    if (this.db) {
      const transaction = this.db.transaction(['assets'], 'readwrite');
      const store = transaction.objectStore('assets');
      await store.put(asset);
    } else {
      const assets = this.getAllAssetsSync();
      assets[asset.id] = asset;
      this.storage.setItem(STORAGE_KEYS.ASSETS, JSON.stringify(assets));
    }
    
    this._notify('asset:created', asset);
    return asset;
  }

  async updateAsset(id, updates) {
    const asset = await this.getAsset(id);
    if (!asset) return null;
    
    const updated = { ...asset, ...updates, updatedAt: new Date().toISOString() };
    return this.saveAsset(updated);
  }

  async deleteAsset(id) {
    if (this.db) {
      const transaction = this.db.transaction(['assets'], 'readwrite');
      const store = transaction.objectStore('assets');
      await store.delete(id);
    } else {
      const assets = this.getAllAssetsSync();
      delete assets[id];
      this.storage.setItem(STORAGE_KEYS.ASSETS, JSON.stringify(assets));
    }
    
    this._notify('asset:deleted', { id });
  }

  async getAsset(id) {
    if (this.db) {
      return new Promise((resolve) => {
        const transaction = this.db.transaction(['assets'], 'readonly');
        const store = transaction.objectStore('assets');
        const request = store.get(id);
        request.onsuccess = () => resolve(request.result);
        request.onerror = () => resolve(null);
      });
    } else {
      const assets = this.getAllAssetsSync();
      return assets[id] || null;
    }
  }

  async getAssets(filters = {}) {
    let assets = Object.values(this.getAllAssetsSync());
    
    if (filters.type) {
      assets = assets.filter(a => a.type === filters.type);
    }
    if (filters.sourceApp) {
      assets = assets.filter(a => a.sourceApp === filters.sourceApp);
    }
    if (filters.search) {
      assets = assets.filter(a => 
        a.title.toLowerCase().includes(filters.search.toLowerCase()) ||
        a.metadata.prompt.toLowerCase().includes(filters.search.toLowerCase())
      );
    }
    
    return assets.sort((a, b) => 
      new Date(b.createdAt) - new Date(a.createdAt)
    );
  }

  getAllAssetsSync() {
    try {
      const data = this.storage.getItem(STORAGE_KEYS.ASSETS);
      return data ? JSON.parse(data) : {};
    } catch {
      return {};
    }
  }

  subscribe(callback) {
    this.subscribers.add(callback);
    return () => this.subscribers.delete(callback);
  }

  _notify(event, data) {
    this.subscribers.forEach(cb => cb(event, data));
    window.dispatchEvent(new CustomEvent(event, { detail: data }));
  }
}

export const assetStore = new AssetStore();
```

### 1.3 assetActions.js - Universal Handoff Actions

```javascript
import { assetStore } from './assetStore.js';
import { ROUTING_TARGETS } from './assetSchema.js';

export async function saveGeneratedAsset(type, data, sourceApp) {
  const asset = await assetStore.saveAsset({
    type,
    sourceApp,
    title: data.title || `Generated ${type}`,
    media: data.media || {},
    metadata: data.metadata || {},

export async function openInDirector(assetId) {
  const asset = await assetStore.getAsset(assetId);
  if (!asset) return;
  
  window.open(`/director?asset=${assetId}`, '_blank');
  assetStore.updateAsset(assetId, {
    'routing.canOpenInDirector': true
  });

export async function openInTimeline(assetId) {

    'routing.canOpenInTimeline': true
  });

export async function openInEditor(assetId) {

export async function openInRender(assetId) {
  const asset = await assetStore.getAsset(assetId);
  if (!asset) return;
  
  window.location.hash = `/render?asset=${assetId}`;
  assetStore.updateAsset(assetId, {
    'routing.canOpenInRender': true

export async function sendToRenderQueue(assetId) {
  const asset = await assetStore.getAsset(assetId);
  if (!asset) return;
  
  const queue = JSON.parse(localStorage.getItem('render_queue') || '[]');
  queue.push({
    id: `render_${Date.now()}`,
    assetId,
    status: 'queued',
    addedAt: new Date().toISOString()
  });
  localStorage.setItem('render_queue', JSON.stringify(queue));
  
  window.dispatchEvent(new CustomEvent('render:queued', { 
    detail: { assetId } 
  }));
}
```

### 1.4 assetEvents.js - Event System

```javascript
export const ASSET_EVENTS = {
  CREATED: 'asset:created',
  UPDATED: 'asset:updated',
  DELETED: 'asset:deleted',
  OPENED: 'asset:opened',
  RENDERED: 'asset:rendered',
  SENT_TO_DIRECTOR: 'asset:sent-to-director',
  SENT_TO_TIMELINE: 'asset:sent-to-timeline',
  SENT_TO_EDITOR: 'asset:sent-to-editor',
  SENT_TO_RENDER: 'asset:sent-to-render'
};

export function createAssetEvent(type, detail) {
  return new CustomEvent(type, { detail });
}

export function listenForAssets(callback) {
  const handleEvent = (event) => callback(event.type, event.detail);
  
  Object.values(ASSET_EVENTS).forEach(eventType => {
    window.addEventListener(eventType, handleEvent);
  });
  
  return () => {
    Object.values(ASSET_EVENTS).forEach(eventType => {
      window.removeEventListener(eventType, handleEvent);
    });
  };
}
```

```javascript
export function createUniversalDropzone(onDrop, acceptTypes = []) {
  const zone = document.createElement('div');
  zone.className = 'universal-dropzone';
  zone.innerHTML = `
    <div class="dropzone-content">
      <div class="drop-icon">📁</div>
      <div class="drop-text">Drop files here</div>
      <div class="drop-subtext">or drag from any app</div>
    </div>
  `;
  
  zone.addEventListener('dragover', (e) => {
    e.preventDefault();
    zone.classList.add('drag-over');
  });
  
  zone.addEventListener('dragleave', () => {
    zone.classList.remove('drag-over');
  });
  
  zone.addEventListener('drop', async (e) => {
    e.preventDefault();
    zone.classList.remove('drag-over');
    
    const files = Array.from(e.dataTransfer.files);
    if (onDrop) await onDrop(files);
  });
  
  return zone;
}
```
