## Phase 3: Pexels API Integration (Priority Task)

### Task 3.1: Create Pexels Provider for MuAPI

**File**: `src/lib/editor/pexelsProvider.js`

**Pattern**: Follow `generationService.js` provider pattern

```javascript
export class PexelsProvider {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseUrl = 'https://api.pexels.com/v1';
    this.videoBaseUrl = 'https://api.pexels.com/videos';
    this.rateLimiter = new RateLimiter({ maxRequests: 200, perMs: 60*60*1000 });
  }

  async searchImages(query, options = {}) {
    await this.rateLimiter.acquire('pexels_search');
    
    const url = `${this.baseUrl}/search?query=${encodeURIComponent(query)}&per_page=${options.per_page || 20}&page=${options.page || 1}`;
    const response = await fetch(url, {
      headers: { 'Authorization': this.apiKey }
    });
    
    if (!response.ok) throw new Error(`Pexels API: ${response.status}`);
    return response.json();
  }

  async searchVideos(query, options = {}) {
    await this.rateLimiter.acquire('pexels_search');
    
    const url = `${this.videoBaseUrl}/search?query=${encodeURIComponent(query)}&per_page=${options.per_page || 20}&page=${options.page || 1}`;
    const response = await fetch(url, {
      headers: { 'Authorization': this.apiKey }
    });
    
    if (!response.ok) throw new Error(`Pexels API: ${response.status}`);
    return response.json();
  }

  async getVideoDetails(videoId) {
    const url = `${this.videoBaseUrl}/${videoId}`;
    const response = await fetch(url, {
      headers: { 'Authorization': this.apiKey }
    });
    return response.json();
  }

  // Transform Pexels data to our asset format
  transformToAsset(pexelsData, type = 'image') {
    return {
      id: `pexels-${pexelsData.id}`,
      type: type,
      name: pexelsData.alt || 'Pexels Media',
      url: type === 'image' ? pexelsData.src.large : pexelsData.video_files.find(f => f.quality === 'hd')?.link || pexelsData.video_files[0].link,
      thumbnail: type === 'image' ? pexelsData.src.tiny : pexelsData.image,
      duration: type === 'video' ? pexelsData.duration : undefined,
      width: pexelsData.width,
      height: pexelsData.height,
      photographer: pexelsData.photographer,
      source: 'pexels'
    };
  }
}
```
